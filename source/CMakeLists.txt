cmake_minimum_required(VERSION 3.20)
project(DeadBabiesXNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(DeadBabiesXNative WIN32 main.cpp frontend.cpp selftest.cpp game.cpp xengine.cpp vendor/stb_vorbis.c)
target_compile_definitions(DeadBabiesXNative PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(DeadBabiesXNative PRIVATE /W4)
else()
  target_compile_options(DeadBabiesXNative PRIVATE -O2 -Wall -Wextra -Wno-unused-parameter -Wno-sign-compare)
  target_link_options(DeadBabiesXNative PRIVATE -static-libgcc -static-libstdc++)
endif()
target_link_libraries(DeadBabiesXNative PRIVATE opengl32 gdi32 user32 ole32 windowscodecs uuid winmm xinput1_4)
